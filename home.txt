  <script>
    toastr.options = {
      "closeButton": false,
      "debug": false,
      "newestOnTop": false,
      "progressBar": true,
      "positionClass": "toast-top-right",
      "preventDuplicates": false,
      "onclick": null,
      "showDuration": "300",
      "hideDuration": "1000",
      "timeOut": "3000",
      "extendedTimeOut": "1000",
      "showEasing": "swing",
      "hideEasing": "linear",
      "showMethod": "fadeIn",
      "hideMethod": "fadeOut"
    };

    $(function () {
      $('#form-login').validate();
      $('#form-login').on('submit', function (e) {
        if ($('#form-login').valid()) {
          $(':input[type="submit"]').prop('disabled', true);
          var formData = new FormData(this);
          $.ajax({
            url: this.action,
            type: 'post',
            data: formData,
            cache: false,
            contentType: false,
            processData: false,
            dataType: 'json',
            success: function (data) {
              if (data.status == 200) {
                toastr.success(data.message);
                window.setTimeout(function () {
                  window.location.href = data.url;
                }, 1000);
              } else {
                $(':input[type="submit"]').prop('disabled', false);
                toastr.error(data.message);
              }
            },
            error: function () {
              $(':input[type="submit"]').prop('disabled', false);
              toastr.error('Something went wrong. Please try again.');
            }
          });
          e.preventDefault();
        }
      });
    });
  </script>
